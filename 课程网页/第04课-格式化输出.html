1printf 与占位符

printf 是什么

和 cout 一样是用来输出的，但要先用一段格式串描述输出长什么样。

占位符

格式串里的 %d 等就是"坑"，后面的参数会按顺序填进去。

头文件

用 printf 要写 #include <cstdio>，它比 cout 更擅长排版。

🧩常用占位符
占位符输出的类型例子
%dint 整数printf("%d", 42); → 42
%lldlong long 长整数printf("%lld", 3000000000LL);
%ffloat / double 小数printf("%f", 3.14); → 3.140000
%c字符 charprintf("%c", 'A'); → A
%s字符串printf("%s", "Hi"); → Hi
%%输出一个百分号 %printf("50%%"); → 50%

⚠ 易错：占位符和参数的类型要对应。%d 对整数、%lld 对 long long、%f 对小数，写错了输出会不正常。

2小数位数与宽度对齐

📐在 % 和字母之间加"修饰"
// 小数位数：.n 表示保留 n 位（会四舍五入）
printf("%.2f", 3.14159);   // 3.14
printf("%.4f", 3.14159);   // 3.1416

// 宽度：数字表示总宽度，不足的补空格
printf("%5d|", 42);          // "   42|"（右对齐）
printf("%-5d|", 42);         // "42   |"（左对齐）
printf("%05d", 42);           // 00042（补 0）

记忆口诀：% 后面先跟对齐（- 左对齐、0 补零），再跟宽度数字，最后可以跟 .小数位数 和类型字母。例如 %08.3f。

3类型转换：让类型对得上

🔄printf 对类型比较"挑剔"
// ❌ double 用 %d 输出：类型不匹配，结果不可靠
double x = 3.5;
printf("%d", x);

// ✅ 先强制转换，把小数截断成整数
printf("%d", (int)x);    // 3

// ✅ 整数用 %f 也可以，会自动变成小数
printf("%f", (double)5);   // 5.000000

⚠ 易错：(int)3.9 是把小数截断成 3，不是四舍五入；四舍五入要靠 %.0f 之类的格式控制。

// printf 基本用法
#include <cstdio>
#include <iostream>
using namespace std;

int main() {
    int a = 10;
    long long b = 3000000000LL;
    double c = 3.14159;
    printf("%d\n", a);      // 整数
    printf("%lld\n", b);    // 长整数
    printf("%f\n", c);      // 小数，默认 6 位
    printf("%c\n", 'A');    // 字符
    return 0;
}
// 控制小数位数
#include <cstdio>

int main() {
    double pi = 3.1415926;
    printf("%.2f\n", pi);   // 3.14
    printf("%.4f\n", pi);   // 3.1416
    printf("%.0f\n", pi);   // 3（会四舍五入）
    return 0;
}
// 宽度与对齐（用 | 当标尺看得更清楚）
#include <cstdio>

int main() {
    printf("%5d|\n", 42);    // "   42|"
    printf("%-5d|\n", 42);   // "42   |"
    printf("%05d\n", 42);     // 00042
    printf("%8.2f\n", 3.5);   // "    3.50"
    return 0;
}
// 把变量填进句子里
#include <cstdio>

int main() {
    int a = 3, b = 5;
    printf("%d + %d = %d\n", a, b, a + b);   // 3 + 5 = 8

    double avg = (a + b) / 2.0;
    printf("平均分：%.2f\n", avg);          // 平均分：4.00
    return 0;
}
// ❌ 占位符和参数类型不匹配
double x = 3.5;
printf("%d", x);        // 类型不匹配，结果不可靠

// ✅ 先转成 int
printf("%d", (int)x);  // 3

// ❌ 忘了 \n，结果全挤在一行
printf("%d%d", 1, 2);

// ✅ 记得换行
printf("%d\n%d\n", 1, 2);

💡 小贴士

  • printf 要头文件：开头写 #include <cstdio>。
  • 占位符按顺序填：格式串里有几个 %，后面就跟几个参数，顺序一一对应。
  • 小数保留位数：用 %.2f 表示两位，%.0f 会四舍五入成整数。
  • 宽度写在字母前：%5d 右对齐，%-5d 左对齐，%05d 补零。
  • 类型要匹配：%d 配 int、%lld 配 long long、%f 配小数。
  • 换行用 \n：printf("%d\n", x); 输出后自动换行。
  • 想输出百分号：写两个 %%。
