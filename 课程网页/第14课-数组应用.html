1数组的四种常见操作

遍历

用一个 for 从下标 0 走到 n-1，把每个元素都看一遍。这是其他操作的基础。

for (int i = 0; i < n; i++) {
    cout << a[i] << " ";
}
求和

准备一个 sum = 0，遍历时把每个元素 sum += a[i]。

int sum = 0;
for (int i = 0; i < n; i++)
    sum += a[i];
最值

先用 a[0] 当前最大，遍历时遇到更大的就更新。

int mx = a[0];
for (int i = 1; i < n; i++)
    if (a[i] > mx) mx = a[i];
查找

想找某个值，就遍历一遍，相等时记下下标；找到就可以 break。

int pos = -1;
for (int i = 0; i < n; i++)
    if (a[i] == target) { pos = i; break; }
套路都一样

这四种操作都是一趟循环 + 一个辅助变量。写熟了，换个题目只是换个更新方式。

⚠ 最值初始化

求最大时初始值别写 0（除非保证都是正数），用 a[0] 最稳；求最小同理。

// 遍历：把每个元素都看一遍
#include <iostream>
using namespace std;

const int N = 1005;
int a[N];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];

    for (int i = 0; i < n; i++) {
        cout << a[i] << " ";
    }
    return 0;
}
// 求和：准备 sum，遍历时累加
#include <iostream>
using namespace std;

const int N = 1005;
int a[N];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];

    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];          // 累加
    }
    cout << sum << endl;
    return 0;
}
// 求最大值：先用 a[0] 当最大
#include <iostream>
using namespace std;

const int N = 1005;
int a[N];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];

    int mx = a[0];
    for (int i = 1; i < n; i++) {
        if (a[i] > mx) mx = a[i];
    }
    cout << mx << endl;
    return 0;
}
// 查找：找了就记下标，并 break
#include <iostream>
using namespace std;

const int N = 1005;
int a[N];

int main() {
    int n, target;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    cin >> target;

    int pos = -1;               // -1 表示没找到
    for (int i = 0; i < n; i++) {
        if (a[i] == target) {
            pos = i;
            break;
        }
    }
    cout << pos << endl;
    return 0;
}
// 综合：一趟遍历同时求 和 / 最大 / 最小
#include <iostream>
using namespace std;

const int N = 1005;
int a[N];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];

    int sum = 0;
    int mx = a[0], mn = a[0];
    for (int i = 0; i < n; i++) {
        sum += a[i];
        if (a[i] > mx) mx = a[i];
        if (a[i] < mn) mn = a[i];
    }
    cout << sum << " " << mx << " " << mn << endl;
    return 0;
}

💡 小贴士

  • 遍历是基本功：for (int i = 0; i < n; i++) 是数组题最常见的开头
  • 求和先清零：int sum = 0; 再在循环里 sum += a[i];
  • 最大初始值用 a[0]：别写 0，免得全是负数时出错；最小同理
  • 查找用 -1 表示没找到：找到就记下标并 break
  • 一趟遍历能同时算好几样：和、最大、最小可以一次循环全求出来
  • 写循环注意边界：是 i < n，不是 i <= n，否则会越界
